Homoglyph visual substitution let deobfuscated = domainBase; for (const [suspect, target] of Object.entries(HOMOGLYPH_MAP)) { deobfuscated = deobfuscated.replaceAll(suspect, target); } if (deobfuscated === brandBase && domainBase !== brandBase) { return { isLookalike: true, impersonatedBrand: brand.name, technique: 'Homoglyph Character Substitution' }; } // 2.
